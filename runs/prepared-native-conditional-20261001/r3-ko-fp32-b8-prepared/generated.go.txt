func ConditionalAssign(input int64) int64 {
	var left = (input + 5)
	var right = (input - 10)
	var guard = (0 < input)
	var guard_alt = (input <= 5)
	if guard {
		left = (input - 10)
	} else {
		right = (input + 5)
	}
	guard_alt = (guard && guard_alt)
	return (left + right)
}
